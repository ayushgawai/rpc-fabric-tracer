cmake_minimum_required(VERSION 3.16)
project(rpc-fabric-tracer CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
include_directories(include)
add_executable(rpc_server src/rpc_server.cpp)
add_executable(loadgen src/loadgen.cpp)
add_executable(linkcheck src/linkcheck.cpp)
enable_testing()
add_executable(test_framing tests/test_framing.cpp)
add_test(NAME framing COMMAND test_framing)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME analyzer COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p "test_analyzer.py" -v)
  set_tests_properties(analyzer PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
